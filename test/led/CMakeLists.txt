cmake_minimum_required(VERSION 3.16)
project(nn2_led_host C)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_executable(nn2_led_tests test_led.c "${ROOT}/components/nn2_board/nn2_board_pca.c")
target_include_directories(nn2_led_tests PRIVATE stubs
    "${ROOT}/components/nn2_board/include" "${ROOT}/components/nn2_core/include"
    "${ROOT}/components/nn2_ports/include" "${ROOT}/main/adapters")
if(MSVC)
    target_compile_options(nn2_led_tests PRIVATE /W4 /WX)
else()
    target_compile_options(nn2_led_tests PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
add_executable(nn2_led_short_tests test_led.c "${ROOT}/components/nn2_board/nn2_board_pca.c")
target_include_directories(nn2_led_short_tests PRIVATE stubs
    "${ROOT}/components/nn2_board/include" "${ROOT}/components/nn2_core/include"
    "${ROOT}/components/nn2_ports/include" "${ROOT}/main/adapters")
target_compile_definitions(nn2_led_short_tests PRIVATE CONFIG_NN2_LED_SHORT_BENCH=1)
if(MSVC)
    target_compile_options(nn2_led_short_tests PRIVATE /W4 /WX)
else()
    target_compile_options(nn2_led_short_tests PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
foreach(t IN ITEMS nn2_led_tests nn2_led_short_tests)
    target_sources(${t} PRIVATE "${ROOT}/components/nn2_app/src/nn2_app_comm.c"
        "${ROOT}/components/nn2_core/src/nn2_core_protocol.c")
    target_include_directories(${t} PRIVATE "${ROOT}/components/nn2_app/include")
endforeach()
enable_testing()
add_test(NAME led_short_owner_and_pca COMMAND nn2_led_short_tests)
add_test(NAME led_owner_and_pca COMMAND nn2_led_tests)
