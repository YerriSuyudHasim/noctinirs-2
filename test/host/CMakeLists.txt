# Product Class U; no IDF roots or conditional platform fallback.
cmake_minimum_required(VERSION 3.16)
project(nn2_host_tests C)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
include("${ROOT}/cmake/nn2_c11.cmake")
# Canonical inventory shared with IDF and the evidence collector.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${ROOT}/tools/layers.json" "${ROOT}/tools/layer_contract.py")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${ROOT}/tools/layer_contract.py" paths
    RESULT_VARIABLE rc OUTPUT_VARIABLE paths OUTPUT_STRIP_TRAILING_WHITESPACE)
if(NOT rc EQUAL 0 OR paths STREQUAL "")
    message(FATAL_ERROR "Invalid or empty canonical layer manifest")
endif()
string(REPLACE "\n" ";" paths "${paths}")
set(INNER_ROOTS)
set(INNER_SOURCES)
set(headers)
foreach(path IN LISTS paths)
    list(APPEND INNER_ROOTS "${ROOT}/${path}/include")
    file(GLOB_RECURSE sources_in_layer CONFIGURE_DEPENDS "${ROOT}/${path}/*.c")
    file(GLOB_RECURSE headers_in_layer CONFIGURE_DEPENDS "${ROOT}/${path}/*.h")
    list(APPEND INNER_SOURCES ${sources_in_layer})
    list(APPEND headers ${headers_in_layer})
endforeach()
add_library(nn2_behavior STATIC ${INNER_SOURCES} "${ROOT}/main/adapters/nn2_ads131_logic.c")

target_include_directories(nn2_behavior PUBLIC ${INNER_ROOTS} "${ROOT}/main/adapters")
target_compile_options(nn2_behavior PRIVATE -Wall -Wextra -Werror -pedantic)
nn2_require_c11(nn2_behavior)
add_library(unity STATIC unity/unity.c)
target_include_directories(unity PUBLIC unity)
# The product and its tests use uint64_t even on a 32-bit host compiler.
target_compile_definitions(unity PUBLIC UNITY_SUPPORT_64)
nn2_require_c11(unity)
add_executable(nn2_host_tests test_main.c test_acquisition.c test_ring.c test_sequencer.c test_contract.c test_storage.c test_ads131.c test_ads131_control.c test_ads131_transport.c test_v4_rtos.c test_v4_sample_clock.c)
target_sources(nn2_host_tests PRIVATE test_v4_sample_edge.c
    "${ROOT}/firmware/v4-idf61/components/nocti_v4_common/nocti_v4_common.c")
target_include_directories(nn2_host_tests PRIVATE v4_sdk_mocks)
target_include_directories(nn2_host_tests PRIVATE "${ROOT}/firmware/v4-idf61/components/nocti_v4_common/include")
target_link_libraries(nn2_host_tests PRIVATE nn2_behavior unity)
target_compile_options(nn2_host_tests PRIVATE -Wall -Wextra -Werror -pedantic)
nn2_require_c11(nn2_host_tests)

# Each header gets fresh preprocessor state. Intentional combined probes follow.
file(WRITE "${CMAKE_BINARY_DIR}/host_headers.tsv" "header\tprobe\n")
foreach(header IN LISTS headers)
    file(RELATIVE_PATH relative "${ROOT}" "${header}")
    string(SHA256 identity "${relative}")
    set(probe "${CMAKE_BINARY_DIR}/header_${identity}.c")
    file(WRITE "${probe}" "#include \"${header}\"\n_Static_assert(sizeof(int) >= 2, \"C11\");\n")
    add_library(header_${identity} OBJECT "${probe}")
    target_include_directories(header_${identity} PRIVATE ${INNER_ROOTS})
    target_compile_options(header_${identity} PRIVATE -Wall -Wextra -Werror -pedantic)
    nn2_require_c11(header_${identity})
    file(APPEND "${CMAKE_BINARY_DIR}/host_headers.tsv" "${relative}\t${probe}\n")
endforeach()
foreach(order IN ITEMS forward reverse)
    set(ordered ${headers})
    if(order STREQUAL "reverse")
        list(REVERSE ordered)
    endif()
    set(probe "${CMAKE_BINARY_DIR}/combined_${order}.c")
    file(WRITE "${probe}" "/* Intentional include-order probe. */\n")
    foreach(header IN LISTS ordered)
        file(APPEND "${probe}" "#include \"${header}\"\n")
    endforeach()
    file(APPEND "${probe}" "_Static_assert(sizeof(int) >= 2, \"C11\");\n")
    add_library(combined_${order} OBJECT "${probe}")
    target_include_directories(combined_${order} PRIVATE ${INNER_ROOTS})
    target_compile_options(combined_${order} PRIVATE -Wall -Wextra -Werror -pedantic)
    nn2_require_c11(combined_${order})
endforeach()
enable_testing()
add_test(NAME nn2_host_tests COMMAND nn2_host_tests)
